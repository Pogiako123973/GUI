function Library:save_flags()
	local flags = HttpService:JSONEncode(Library.Flags)

	writefile('Aries/' .. game.GameId .. '.lua', flags)
end

function Library:load_flags()
	if not isfile('Aries/' .. game.GameId .. '.lua') then
		Library.save_flags()
		return
	end

	local flags = readfile('Aries/' .. game.GameId .. '.lua')

	if not flags then
		Library.save_flags()
		return
	end

	Library.Flags = HttpService:JSONDecode(flags)
end

Library.load_flags()

function Library:save_preset(name, data)
	local encoded = HttpService:JSONEncode(data)
	writefile("Aries/Presets/" .. name .. ".json", encoded)
end

function Library:load_preset(name)
	local path = "Aries/Presets/" .. name .. ".json"

	if not isfile(path) then
		return nil
	end

	local decoded = HttpService:JSONDecode(readfile(path))
	return decoded
end

-- Auto-load of last preset on startup removed; presets are only displayed in the UI

function Library:list_presets()
	local files = listfiles("Aries/Presets")
	local presets = {}

	for _, file in files do
		local presetName = file:match("Aries[/\\]Presets[/\\](.+)%.json")

		if presetName then
			table.insert(presets, presetName)
		end
	end

	return presets
end

function Library:validate(defaults, options)
	options = options or {}

	for Index, Value in defaults do
		if options[Index] == nil then
			options[Index] = Value
		end
	end

	return options
end

type Icon = {
	Url: string,
	Id: number,
	IconName: string,
	ImageRectOffset: Vector2,
	ImageRectSize: Vector2,
}

type IconModule = {
	Icons: { string },
	GetAsset: (Name: string) -> Icon?,
}

local FetchIcons, Icons = pcall(function()
	return (loadstring(
		game:HttpGet("https://raw.githubusercontent.com/mstudio45/lucide-roblox-direct/refs/heads/main/source.lua")
		) :: () -> IconModule)()
end)

local function IsCustomAssetIcon(Icon: string, IncludeAssetId: boolean)
	return typeof(Icon) == "string" and (Icon:match("^content://") or Icon:match("^rbxasset://%x+/") or (IncludeAssetId == true and Icon:match("^rbxassetid://")))
end

function IsValidCustomIcon(Icon: string)
	return typeof(Icon) == "string" and (Icon:match("^rbxasset://textures/") or Icon:match("roblox%.com/asset/%?id=") or Icon:match("rbxthumb://type="))
end

function Library:GetIcon(IconName: string)
	if not FetchIcons or not Icons then
		return
	end

	local Success, Icon = pcall(Icons.GetAsset, IconName)
	if not Success then
		return
	end

	return Icon
end

function Library:GetCustomIcon(IconName: string): any
	if not IconName then
		return nil
	end

	if tonumber(IconName) then
		IconName = string.format("rbxassetid://%s", tostring(IconName))
	end

	if IsCustomAssetIcon(IconName, true) then
		return {
			Url = IconName,
			ImageRectOffset = Vector2.zero,
			ImageRectSize = Vector2.zero,
		}
	elseif IsValidCustomIcon(IconName) then
		return {
			Url = IconName,
			ImageRectOffset = Vector2.zero,
			ImageRectSize = Vector2.zero,
			Custom = true,
		}
	end

	local LucideIcon = Library:GetIcon(IconName)
	if LucideIcon then
		return LucideIcon
	end

	return nil
end

	-- Keybind picker logic
	local function getKeyName(input)
		if input.UserInputType == Enum.UserInputType.Keyboard then
			return input.KeyCode
		elseif input.UserInputType == Enum.UserInputType.MouseButton1 then
			return Enum.UserInputType.MouseButton1
		elseif input.UserInputType == Enum.UserInputType.MouseButton2 then
			return Enum.UserInputType.MouseButton2
		elseif input.UserInputType == Enum.UserInputType.MouseButton3 then
			return Enum.UserInputType.MouseButton3
		end

		return nil
	end

	local function formatKeyName(key)
		if typeof(key) == "EnumItem" then
			if key.EnumType == Enum.UserInputType then
				if key == Enum.UserInputType.MouseButton1 then return "M1" end
				if key == Enum.UserInputType.MouseButton2 then return "M2" end
				if key == Enum.UserInputType.MouseButton3 then return "M3" end
			end
			local name = key.Name
			name = name:gsub("Left", "L"):gsub("Right", "R"):gsub("Control", "Ctrl"):gsub("Shift", "Shf")
			return name
		end
		return "None"
	end

		if not GUI.WaitingForKey then return end

		-- Escape clears the keybind
		if input.KeyCode == Enum.KeyCode.End then
			GUI.WaitingForKey = false
			local toggle = GUI.ActiveKeybindToggle
			if toggle then
				toggle.Keybind = nil
				toggle.KeybindName = nil
				Library.Keybinds[toggle] = nil
				if toggle.flag then
					Library.Flags[toggle.flag .. "_keybind"] = nil
					Library.save_flags()
				end
				if toggle["72_text"] then
					toggle["72_text"].Text = "···"
					toggle["72_text"].TextColor3 = Color3.fromRGB(138, 138, 148)
				end
				if toggle["73_s"] then
					TweenService:Create(toggle["73_s"], TweenInfo.new(0.2), {Transparency = 1}):Play()
				end
			end
			GUI["kpw_nbf_p"].Text = "New bind"
			GUI["kpw_nbf_p"].TextColor3 = Color3.fromRGB(138, 138, 148)
			TweenService:Create(GUI["kpw_nbf_s"], TweenInfo.new(0.2), {Transparency = 1}):Play()
			return
		end

		local key = getKeyName(input)

		if key == nil then
			return
		end

		if table.find(Ignored_Keys, key) then
			return
		end

		GUI.WaitingForKey = false

		local toggle = GUI.ActiveKeybindToggle
		if not toggle then return end

		-- Set the keybind
		toggle.Keybind = key
		toggle.KeybindName = formatKeyName(key)
		if not toggle.KeybindMode then
			toggle.KeybindMode = "toggle"
		end

		-- Register in global keybinds list
		Library.Keybinds[toggle] = {
			key = key,
			mode = toggle.KeybindMode,
			flag = toggle.flag
		}

		-- Save keybind to flags
		if toggle.flag then
			Library.Flags[toggle.flag .. "_keybind"] = {
				keyName = key.Name,
				keyType = key.EnumType == Enum.KeyCode and "KeyCode" or "UserInputType",
				displayName = formatKeyName(key),
				mode = toggle.KeybindMode
			}
			Library.save_flags()
		end

		-- Update UI
		GUI["kpw_nbf_p"].Text = formatKeyName(key)
		GUI["kpw_nbf_p"].TextColor3 = Color3.fromRGB(243, 243, 247)
		TweenService:Create(GUI["kpw_nbf_s"], TweenInfo.new(0.2), {Transparency = 1}):Play()

		if toggle["72_text"] then
			toggle["72_text"].Text = formatKeyName(key)
			toggle["72_text"].TextColor3 = Color3.fromRGB(125, 109, 247)
		end

		if toggle["73_s"] then
			TweenService:Create(toggle["73_s"], TweenInfo.new(0.2), {Transparency = 0}):Play()
		end
	end)

	local MovementKeys = {
		[Enum.KeyCode.W] = true,
		[Enum.KeyCode.A] = true,
		[Enum.KeyCode.S] = true,
		[Enum.KeyCode.D] = true,

		[Enum.KeyCode.Space] = true,

		[Enum.KeyCode.Down] = true,

	UserInputService.InputBegan:Connect(function(input, gpe)
		if GUI.WaitingForKey then return end

		local key = getKeyName(input)
		if key == nil then return end

		if gpe then
			return
		end

		if input.UserInputType == Enum.UserInputType.Keyboard then
			if MovementKeys[input.KeyCode] then
				return
			end
		end

		for toggle, bind in pairs(Library.Keybinds) do
			if bind.key == key and toggle.flag then
				if bind.mode == "toggle" then
					-- Only trigger on the actual key press, not repeats
					if not toggle._keybindHeld then
						toggle._keybindHeld = true
						pcall(function() toggle:Toggle() end)
					end
				elseif bind.mode == "hold" then
					toggle._keybindHeld = true
					pcall(function() toggle:Toggle(true) end)
				end
			end
		end
	end)

	UserInputService.InputEnded:Connect(function(input, gpe)
		local key = getKeyName(input)
		if key == nil then return end

		for toggle, bind in pairs(Library.Keybinds) do
			if bind.key == key and toggle.flag then
				if bind.mode == "toggle" then
					toggle._keybindHeld = false
				elseif bind.mode == "hold" then
					toggle._keybindHeld = false
					pcall(function() toggle:Toggle(false) end)
				end
			end
		end
	end)

	local function cpRender()
		local color = cpCurrentColor()

		TweenService:Create(GUI["cpw_pal"], TweenInfo.new(0.15, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
			BackgroundColor3 = Color3.fromHSV(cpState.hue, 1, 1)
		}):Play()
		TweenService:Create(GUI["cpw_pal_sel"], TweenInfo.new(0.15, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
			Position = UDim2.new(math.min(cpState.saturation, 0.92), 0, math.min(1 - cpState.value, 0.92), 0)
		}):Play()
		TweenService:Create(GUI["cpw_hue_k"], TweenInfo.new(0.15, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
			Position = UDim2.new(0, 0, math.min(cpState.hue, 0.92), 0)
		}):Play()
		TweenService:Create(GUI["cpw_alpha_k"], TweenInfo.new(0.15, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
			Position = UDim2.new(math.min(cpState.alpha, 0.93), 0, 0, 0)
		}):Play()
		TweenService:Create(GUI["cpw_alpha_ramp"], TweenInfo.new(0.15, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
			BackgroundColor3 = color
		}):Play()

		GUI["cpw_hex_v"].Text = cpCurrentHex()

		if cpState.currentTarget then
			TweenService:Create(cpState.currentTarget.swatch, TweenInfo.new(0.15, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
				BackgroundColor3 = color
			}):Play()
			if cpState.currentTarget.callback then
				cpState.currentTarget.callback(color)
			end

	local ARROW_GAP = 2   -- gap after section text before arrow
	local ARROW_WIDTH = 20
	local ELEMENT_GAP = 4  -- gap after arrow before element text

	local function updateBreadcrumbLayout()
		-- TextBounds returns already-scaled pixel values, but these offsets are
		-- applied inside the UIScale frame, so they'd be scaled again. Dividing
		-- by the target uiScale compensates for the double-scaling.
		-- We use _targetScale (not the live Scale) because the UIScale tweens
		-- from 0 on open, and using the live value during the tween would
		-- cause division by zero or incorrect positions.
		local uiScale = GUI._targetScale or 1
		if uiScale <= 0 then uiScale = 1 end

		local sectionX = ICON_X + ICON_SIZE + SECTION_GAP
		local sectionWidth = math.max(math.ceil(BreadcrumbSection.TextBounds.X / uiScale), 1)
		BreadcrumbSection.Size = UDim2.new(0, sectionWidth, 0, 22)
		BreadcrumbSection.Position = UDim2.new(0, sectionX, 0, 0)

		local arrowX = sectionX + sectionWidth + ARROW_GAP
		BreadcrumbArrow.Position = UDim2.new(0, arrowX, 0, -4)

		local elementX = arrowX + ARROW_WIDTH + ELEMENT_GAP
		local elementWidth = math.max(math.ceil(BreadcrumbElement.TextBounds.X / uiScale), 1)
		BreadcrumbElement.Size = UDim2.new(0, elementWidth, 0, 22)
		BreadcrumbElement.Position = UDim2.new(0, elementX, 0, 0)
	end

	BreadcrumbSection:GetPropertyChangedSignal("TextBounds"):Connect(updateBreadcrumbLayout)
	BreadcrumbElement:GetPropertyChangedSignal("TextBounds"):Connect(updateBreadcrumbLayout)

			local label = child:FindFirstChild("Label", true)

				-- Restore normal behavior
				SearchPlaceholder.Visible = true
				searchScrollLeft.Visible = false
				searchScrollRight.Visible = false

				for _, tab in (GUI.Tabs) do
					local containers = tab.AllContainers or {{container = tab.ContentContainer, section = tab.Section or 'left'}}
					for _, entry in (containers) do
						-- Restore parent to main window if it was reparented to a scroll frame
						if entry.container.Parent ~= GUI["2"] then
							entry.container.Parent = GUI["2"]

						for _, child in (entry.container:GetChildren()) do

					end
				end
			else
				SearchPlaceholder.Visible = false
				searchScrollLeft.Visible = true
				searchScrollRight.Visible = true

				local leftIndex = 0
				local rightIndex = 0

				for _, tab in (GUI.Tabs) do
					local containers = tab.AllContainers or {{container = tab.ContentContainer, section = tab.Section or 'left'}}
					for _, entry in (containers) do
						local hasMatch = false
						for _, child in (entry.container:GetChildren()) do
							if child:IsA("GuiObject") then
								local widgetName = getWidgetName(child)
								if string.find(string.lower(widgetName), query) then
									child.Visible = true
									hasMatch = true
								else
									child.Visible = false
								end
							end
						end
						if hasMatch then
							-- Reparent to scroll frame and let UIListLayout handle stacking
							if entry.section == 'right' then
								entry.container.Parent = searchScrollRight
								entry.container.LayoutOrder = rightIndex
								rightIndex = rightIndex + 1
							else
								entry.container.Parent = searchScrollLeft
								entry.container.LayoutOrder = leftIndex
								leftIndex = leftIndex + 1
							end
							entry.container.Position = UDim2.new(0, 0, 0, 0)
							entry.container.Visible = true
						else
							-- Hide and restore parent to main window
							entry.container.Visible = false
							if entry.container.Parent ~= GUI["2"] then
								entry.container.Parent = GUI["2"]
							end
						end
					end
				end
			end
		end)
	end

		local function snapshotFlags()
			local out = {}
			for k, v in Library.Flags do
				if type(v) == "table" then
					local copy = {}
					for k2, v2 in v do
						copy[k2] = v2
					end
					out[k] = copy
				else
					out[k] = v
				end
			end
			return out
		end

		local function applyFlags(flags)
			if not flags then return end
			-- Clear existing flags except __lastPreset so each preset is independent
			for k in pairs(Library.Flags) do
				if k ~= "__lastPreset" then
					Library.Flags[k] = nil
				end
			end
			-- Apply the new preset's flags
			for k, v in flags do
				Library.Flags[k] = v
			end
		end

		local function serializeFlags(flags)
			local out = {}
			for k, v in pairs(flags) do
				if k == "__lastPreset" then continue end
				if typeof(v) == "Color3" then
					out[k] = {__type = "Color3", R = v.R, G = v.G, B = v.B}
				else
					out[k] = v
				end
			end
			return out
		end

		local function deserializeFlags(data)
			if not data then return {} end
			local out = {}
			for k, v in pairs(data) do
				if type(v) == "table" and v.__type == "Color3" then
					out[k] = Color3.new(v.R or 0, v.G or 0, v.B or 0)
				elseif type(v) == "table" and v.R ~= nil and v.G ~= nil and v.B ~= nil and v.A == nil and v.__type == nil and v.keyName == nil then
					out[k] = Color3.new(v.R, v.G, v.B)
				else
					out[k] = v
				end
			end
			return out
		end

		local function scheduleSave()
			if loading then return end

			for _, preset in presets do

				Library:save_preset(preset.name, data)

				Library.Flags.__lastPreset = activePreset.name
			else
				Library.Flags.__lastPreset = nil

		local function selectPreset(preset)
			if activePreset == preset then return end

				activePreset.flags = snapshotFlags()
			end

			GUI.ApplyingPreset = true

			-- Phase 1: Turn off all toggle flags so nothing from the old preset interacts with the new one
			for k, v in pairs(Library.Flags) do
				if k ~= "__lastPreset" and type(v) == "boolean" then
					Library.Flags[k] = false
				end
			end
			for _, refresh in ipairs(GUI.Widgets) do
				refresh()
			end

			-- Phase 2: Apply the new preset's flags and refresh all widgets
			applyFlags(preset.flags)
			for _, refresh in ipairs(GUI.Widgets) do
				refresh()
			end

			GUI.ApplyingPreset = false

			scheduleSave()

					addPreset(name, flags, false)
				end
			end

		else

			for index, name in DEFAULT_PRESETS do
				addPreset(name, nil, index == 1)
			end
		end

			renderRows()
		end)

			sortAscending = not sortAscending

				SortIcon,
				TweenInfo.new(0.2, Enum.EasingStyle.Quint, Enum.EasingDirection.Out),
				{
					ImageColor3 = sortAscending and COLOR_SORT_ACTIVE or COLOR_SORT_IDLE
				}
			):Play()

			renderRows()
		end)

		AddHit.MouseButton1Click:Connect(function()
			local baseName = "Config " .. tostring(#presets + 1)
			local newName  = baseName
			local counter  = 1

			local function nameTaken(n)
				for _, p in presets do
					if p.name == n then return true end
				end
				return false
			end

			while nameTaken(newName) do
				counter += 1
				newName = baseName .. " (" .. counter .. ")"
			end

			local preset = addPreset(newName, snapshotFlags(), true)
			selectPreset(preset)
		end)

		renderRows()
		renderStates()
	end

	return GUI

	function Tab:Activate()
		if not Tab.Active then
			local sections = {}
			for _, entry in (Tab.AllContainers) do
				sections[entry.section] = true
			end
			for section, _ in pairs(sections) do
				if GUI.CurrentTabs[section] ~= nil and GUI.CurrentTabs[section] ~= Tab then
					GUI.CurrentTabs[section]:Deactivate()
				end
			end

			Tab.Active = true
			for _, entry in (Tab.AllContainers) do
				GUI.CurrentTabs[entry.section] = Tab
				entry.container.Visible = true
			end
		end
	end

	function Tab:Deactivate()
		if Tab.Active then
			Tab.Active = false
			Tab.Hover = false
			for _, entry in (Tab.AllContainers) do
				entry.container.Visible = false
			end
		end
	end

		function Toggle:Toggle(state)
			if Library.Flags[self.flag] == nil then
				Library.Flags[self.flag] = false
			end

			if state == nil then
				Library.Flags[self.flag] = not Library.Flags[self.flag]
			else
				Library.Flags[self.flag] = state
			end

			if self._activeTweens then
				for _, tween in self._activeTweens do
					tween:Cancel()
				end
			end
			self._activeTweens = {}

			local bgFrame = Toggle["74"]
			local knob = Toggle["76"]
			if not bgFrame or not knob then return end

			if Library.Flags[self.flag] then
				local t1 = TweenService:Create(bgFrame, TweenInfo.new(0.2), { BackgroundColor3 = Color3.fromRGB(125, 109, 247) })
				local t2 = TweenService:Create(knob, TweenInfo.new(0.2), { Position = UDim2.new(0, 15, 0, 2), BackgroundColor3 = Color3.fromRGB(255, 255, 255) })
				t1:Play()
				t2:Play()
				table.insert(self._activeTweens, t1)
				table.insert(self._activeTweens, t2)
			else
				local t1 = TweenService:Create(bgFrame, TweenInfo.new(0.2), { BackgroundColor3 = Color3.fromRGB(37, 37, 47) })
				local t2 = TweenService:Create(knob, TweenInfo.new(0.2), { Position = UDim2.new(0, 2, 0, 2), BackgroundColor3 = Color3.fromRGB(64, 64, 76) })
				t1:Play()
				t2:Play()
				table.insert(self._activeTweens, t1)
				table.insert(self._activeTweens, t2)
			end

			if not GUI.ApplyingPreset then
				pcall(Library.save_flags)
			end

			if options.callback then
				options.callback(Library.Flags[self.flag])
			end
		end

		if Library.Flags[Toggle.flag] == nil then
			Library.Flags[Toggle.flag] = false
		end

			if v == nil then
				local mousePos = inputPos or UserInputService:GetMouseLocation()
				local Percentage = math.clamp((mousePos.X - Slider['7e'].AbsolutePosition.X) / (Slider['7e'].AbsoluteSize.X), 0, 1)
				local Value = math.round(((options.max - options.min) * Percentage) + options.min)

				Slider['7d'].Text = tostring(Value) .. '%'
				Slider["82"]["Position"] = UDim2.new(Percentage, 0, 0.5, 0)

				Slider["80"].Size = UDim2.fromScale(Percentage, 1)
			else
				Slider['7d'].Text = tostring(v) .. '%'
				local pct = (v - options.min) / (options.max - options.min)
				Slider['80'].Size = UDim2.fromScale(pct, 1)
				Slider["82"]["Position"] = UDim2.new(pct, 0, 0.5, 0)
			end

			if Slider.flag then
				Library.Flags[Slider.flag] = Slider:GetValue()
				if not GUI.ApplyingPreset then
					Library.save_flags()
				end
			end

			if options.callback then
				options.callback(Slider:GetValue())
			end
		end

			local text = Slider['7d'].Text
			local num = tonumber(text:match("%-?%d+"))
			return num
		end

			if sliderDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then

			end
		end)

		UserInputService.InputEnded:Connect(function(input, gpe)
			if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
				sliderDragging = false
			end
		end)

			if Library.Flags[Slider.flag] ~= nil then
				Slider:SetValue(Library.Flags[Slider.flag])

				end

		-- Open dropdown popup
		local function openDropdownPopup()
			local popup = GUI.DropdownPopup
			local anchor = Dropdown["d72"]

			-- If already open for this dropdown, close it
			if popup.Visible and GUI.DropdownPopupData.currentAnchor == anchor then
				GUI.closeDropdownPopup()
				return
			end

			-- Close any existing popup instantly
			if popup.Visible then
				popup.Visible = false
				if GUI.DropdownPopupData.currentDropdown then
					GUI.DropdownPopupData.currentDropdown.Open = false
					GUI.DropdownPopupData.currentDropdown["d74"].Rotation = 0
				end
			end

			Dropdown.Open = true
			GUI.DropdownPopupData.currentAnchor = anchor
			GUI.DropdownPopupData.currentCallback = function(value)
				Dropdown.Selected = value
				Dropdown["d73"].Text = value
				if Dropdown.flag then
					Library.Flags[Dropdown.flag] = value
					Library.save_flags()
				end
				if options.callback then
					options.callback(value)
				end
			end
			GUI.DropdownPopupData.currentDropdown = Dropdown

			-- Rotate caret (Winds UI style)
			TweenService:Create(Dropdown["d74"], TweenInfo.new(0.2, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
				Rotation = 180
			}):Play()

			-- Clear old option frames
			for _, child in popup:GetChildren() do
				if (child:IsA("Frame") or child:IsA("TextButton")) and child.Name ~= "DragHandle" then
					child:Destroy()
				end
			end

			-- Calculate width based on longest option (Winds UI style)
			local ROW_HEIGHT = 26
			local MIN_WIDTH = 139
			local MAX_WIDTH = 240
			local width = MIN_WIDTH

			for _, option in options.options do
				local charWidth = #option * 7
				width = math.max(width, charWidth + 54)
			end
			width = math.min(width, MAX_WIDTH)

			-- Create option rows
			local rows = {}

	function Tab:Create_Section(name, side)
		-- Backwards compatibility:
		--   Old: Create_Section('left')         -> first arg is side
		--   New: Create_Section('Title', 'left') -> name + side
		--   New: Create_Section('Title')        -> name only, side defaults to tab's section
		local sectionName, sectionSide

		if name == 'left' or name == 'right' then
			sectionSide = name
			sectionName = nil
		else
			sectionName = name
			sectionSide = side or Tab.Section or 'left'
		end

		local Section = {
			section = sectionSide,
			Tab = Tab,
			title = sectionName
		}

		-- Get or create a column wrapper for vertical stacking of sections
		local wrapperKey = "_wrapper_" .. sectionSide
		if not Tab[wrapperKey] then
			Tab[wrapperKey] = Instance.new("Frame", GUI["2"])
			Tab[wrapperKey].BorderSizePixel = 0
			Tab[wrapperKey].BackgroundTransparency = 1
			Tab[wrapperKey].AutomaticSize = Enum.AutomaticSize.Y
			Tab[wrapperKey].Size = UDim2.new(0, 264, 0, 0)
			Tab[wrapperKey].Position = UDim2.new(0, sectionSide == 'right' and 444 or 166, 0, 118)
			Tab[wrapperKey].Name = (Tab["12"].Name or "Tab") .. "_" .. sectionSide .. "_wrapper"

			local wrapperLayout = Instance.new("UIListLayout", Tab[wrapperKey])
			wrapperLayout.SortOrder = Enum.SortOrder.LayoutOrder
			wrapperLayout.Padding = UDim.new(0, 8)

			if not GUI.CurrentTabs[sectionSide] then
				GUI.CurrentTabs[sectionSide] = Tab
			end

			Tab[wrapperKey].Visible = (GUI.CurrentTabs[sectionSide] == Tab)

			table.insert(Tab.AllContainers, {container = Tab[wrapperKey], section = sectionSide})
		end

		local wrapper = Tab[wrapperKey]

		-- Create per-section MainTitle if a name is provided
		if sectionName then
			Section.MainTitle = Instance.new("TextLabel", wrapper)
			Section.MainTitle.TextSize = 13
			Section.MainTitle.TextXAlignment = Enum.TextXAlignment.Left
			Section.MainTitle.FontFace = Font.new("rbxasset://fonts/families/Montserrat.json", Enum.FontWeight.SemiBold, Enum.FontStyle.Normal)
			Section.MainTitle.TextColor3 = Color3.fromRGB(243, 243, 247)
			Section.MainTitle.BackgroundTransparency = 1

		Section.ContentContainer.Visible = true  -- Visibility handled by wrapper

		sectionCorner.CornerRadius = UDim.new(0, 10)

		local sectionShadow = Instance.new("UIShadow", Section.ContentContainer)
		sectionShadow:SetAttribute("fadeOriginal_Transparency", 0.55)

		local sectionLayout = Instance.new("UIListLayout", Section.ContentContainer)
		sectionLayout.SortOrder = Enum.SortOrder.LayoutOrder

		function Section:Create_Toggle(opts)
			opts = opts or {}
			opts.tab = Section
			return Tab:Create_Toggle(opts)
		end

		function Section:Create_Slider(opts)

		end

		end

		function Section:Create_ColorPicker(opts)

		return Section
	end

	return Tab
end

	dur = dur or 4

	local UI_Notification = {}

	UI_Notification["1"] = Instance.new("ScreenGui")

	UI_Notification["1"].ZIndexBehavior = Enum.ZIndexBehavior.Sibling

	UI_Notification["2"].AnchorPoint = Vector2.new(1, 1)

	Layout.Padding = UDim.new(0, 8)

	-- Template (your original UI_Notification["4"])
	UI_Notification["4"] = Instance.new("Frame")
	UI_Notification["4"].Size = UDim2.fromOffset(230, 52)
	UI_Notification["4"].BackgroundTransparency = 1
	UI_Notification["4"].ZIndex = 10

	Icon_Frame.BackgroundColor3 = Color3.fromRGB(47, 47, 50)

	local Icon_Label = Instance.new("TextLabel", Icon_Frame)
	Icon_Label.BackgroundTransparency = 1
	Icon_Label.Size = UDim2.fromScale(1, 1)
	Icon_Label.Font = Enum.Font.GothamBold
	Icon_Label.Text = "i"
	Icon_Label.TextSize = 12
	Icon_Label.TextColor3 = Color3.fromRGB(236, 236, 236)
	Icon_Label.ZIndex = 12

	local Title_Label = Instance.new("TextLabel", Card)
	Title_Label.BackgroundTransparency = 1
	Title_Label.Position = UDim2.fromOffset(46, 12)

	Title_Label.Font = Enum.Font.GothamBold

	Title_Label.ZIndex = 11

	Track.Name = "track"
	Track.Position = UDim2.fromOffset(46, 34)
	Track.Size = UDim2.new(1, -60, 0, 3)
	Track.BackgroundColor3 = Color3.fromRGB(56, 56, 59)
	Track.BorderSizePixel = 0
	Track.ZIndex = 11
	Instance.new("UICorner", Track).CornerRadius = UDim.new(0, 2)

	local Fill = Instance.new("Frame", Track)
	Fill.Name = "fill"
	Fill.Size = UDim2.fromScale(1, 1)
	Fill.BackgroundColor3 = Color3.fromRGB(236, 236, 236)
	Fill.BorderSizePixel = 0
	Fill.ZIndex = 12

	Instance.new("UICorner", Fill).CornerRadius = UDim.new(0, 2)

	-- Clone template
	local Notification = UI_Notification["4"]:Clone()
	Notification.Parent = UI_Notification["2"]
	Notification.Visible = true

	local Card_Clone = Notification:FindFirstChildOfClass("Frame")
	local Track_Clone = Card_Clone:FindFirstChild("track")
	local Fill_Clone = Track_Clone and Track_Clone:FindFirstChild("fill")

	-- Animate bar drain
	if Fill_Clone then
		game:GetService("TweenService"):Create(
			Fill_Clone,
			TweenInfo.new(dur, Enum.EasingStyle.Linear),
			{ Size = UDim2.new(0, 0, 1, 0) }
		):Play()
	end

	-- Slide-in animation
	game:GetService("TweenService"):Create(
		Card_Clone,
		TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
		{ Position = UDim2.fromOffset(0, 0) }
	):Play()

	-- Slide-out + collapse
	task.delay(dur, function()
		local out = game:GetService("TweenService"):Create(
			Card_Clone,
			TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
			{ Position = UDim2.fromOffset(260, 0) }
		)
		out:Play()

		out.Completed:Once(function()
			local collapse = game:GetService("TweenService"):Create(
				Notification,
				TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
				{ Size = UDim2.fromOffset(230, 0) }
			)
			collapse:Play()

			collapse.Completed:Once(function()
				Notification:Destroy()
			end)
		end)
	end)
end
